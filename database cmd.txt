-- ============================================================
-- FOODIEHUB DATABASE CHECKING COMMANDS
-- Purpose:
-- These commands are only for viewing and verifying data.
-- They do NOT INSERT, UPDATE, or DELETE anything.
-- ============================================================


-- ============================================================
-- 1. SELECT THE DATABASE
-- ============================================================
-- Tells MySQL that we want to work with the foodiehub database.

USE foodiehub;


-- ============================================================
-- 2. CHECK WHICH DATABASE IS CURRENTLY SELECTED
-- ============================================================
-- Shows the database currently being used.

SELECT DATABASE();


-- ============================================================
-- 3. SHOW ALL TABLES
-- ============================================================
-- Displays all tables available inside the foodiehub database.

SHOW TABLES;


-- ============================================================
-- 4. CHECK ALL RESTAURANTS
-- ============================================================
-- Shows restaurant ID, name, rating, location and image.
-- Useful for checking restaurant details.

SELECT
    id,
    name,
    rating,
    location,
    image
FROM crisp_restaurant
ORDER BY id;


-- ============================================================
-- 5. CHECK ALL FOOD / MENU ITEMS
-- ============================================================
-- Shows every food item with its restaurant ID.
-- restaurant_id tells us which restaurant owns the food.

SELECT
    id,
    name,
    price,
    restaurant_id,
    image
FROM crisp_food
ORDER BY restaurant_id, id;


-- ============================================================
-- 6. CHECK HOW MANY RESTAURANTS EXIST

-- ============================================================
-- 7. CHECK HOW MANY FOOD ITEMS EXIST
-- ============================================================
-- Expected result in your current project: 54

SELECT COUNT(*) AS total_foods
FROM crisp_food;

-- ============================================================
-- 19. CHECK FOR DUPLICATE RESTAURANT NAMES
-- ============================================================
-- Finds restaurants with the same name more than once.
-- Expected result: Empty set

SELECT
    name,
    COUNT(*) AS total
FROM crisp_restaurant
GROUP BY name
HAVING COUNT(*) > 1;


-- ============================================================
-- 20. CHECK FOR DUPLICATE FOOD NAMES
-- ============================================================
-- Checks whether the same food exists more than once
-- inside the same restaurant.
-- Expected result: Empty set

SELECT
    restaurant_id,
    name,
    COUNT(*) AS total
FROM crisp_food
GROUP BY restaurant_id, name
HAVING COUNT(*) > 1;


-- ============================================================
-- 21. CHECK RESTAURANT + FOOD RELATIONSHIP
-- ============================================================
-- Joins both tables so you can clearly see:
-- Food name + Restaurant name + Price.

SELECT
    f.id AS food_id,
    f.name AS food_name,
    f.price,
    r.id AS restaurant_id,
    r.name AS restaurant_name
FROM crisp_food f
JOIN crisp_restaurant r
    ON f.restaurant_id = r.id
ORDER BY r.id, f.id;


-- ============================================================
-- 22. CHECK HOW MANY FOODS EACH RESTAURANT HAS
-- ============================================================
-- Shows the number of menu items belonging to each restaurant.

SELECT
    r.id AS restaurant_id,
    r.name AS restaurant_name,
    COUNT(f.id) AS total_foods
FROM crisp_restaurant r
LEFT JOIN crisp_food f
    ON r.id = f.restaurant_id
GROUP BY r.id, r.name
ORDER BY r.id;


-- ============================================================
-- 23. CHECK FOR FOODS WITHOUT A VALID RESTAURANT
-- ============================================================
-- Finds food records whose restaurant relationship
-- does not match a restaurant.
-- Normally this should return Empty set.

SELECT
    f.id,
    f.name,
    f.restaurant_id
FROM crisp_food f
LEFT JOIN crisp_restaurant r
    ON f.restaurant_id = r.id
WHERE r.id IS NULL;
